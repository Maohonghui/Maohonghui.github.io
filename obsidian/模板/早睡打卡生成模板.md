<%*
/*
 * 早睡打卡 —— 生成/补全一整年的打卡清单
 * ════════════════════════════════════════════════════════════════════════
 *
 * 这个模板做的事：往当前笔记里插入从今天到年底的打卡行，
 * 每行长这样：   - [ ] 2026-10-09
 *
 * 已经存在的行不会重复插入，所以你可以放心反复运行它。
 *
 * 装法：放到你 Obsidian 库的模板文件夹里（跟博客文章模板放一起）。
 * 用法：打开「早睡打卡」这篇笔记 → 命令面板 → Templater: Replace templates in
 *       the active file
 *
 * 打勾之后那行会变成 - [x] 2026-10-09，博客就是靠这个生成热力图的。
 */
const pad = (n) => String(n).padStart(2, '0');
const fmt = (d) => `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`;

/* 读当前笔记内容，看看哪些日期已经写过了 */
let existing = '';
try {
  existing = await tp.file.content;
} catch (e) {
  existing = '';
}

const done = new Set();
const re = /^[-\s]*\[[ xX]\][\s\S]{0,3}(\d{4}-\d{2}-\d{2})/gm;
let m;
while ((m = re.exec(existing)) !== null) {
  done.add(m[1]);
}

/* 从今天生成到今年年底 */
const today = new Date();
const endOfYear = new Date(today.getFullYear(), 11, 31);
const lines = [];

const cursor = new Date(today.getFullYear(), today.getMonth(), today.getDate());
while (cursor <= endOfYear) {
  const day = fmt(cursor);
  if (!done.has(day)) {
    const isToday = day === fmt(today);
    lines.push(`- [ ] ${day}${isToday ? '   ← 今天' : ''}`);
  }
  cursor.setDate(cursor.getDate() + 1);
}

/* 已经写全了就不用再插 */
if (lines.length === 0) {
  tR += '\n> 今年剩下的日子都已经在清单里了，不需要重新生成。\n';
} else {
  tR += '\n' + lines.join('\n') + '\n';
}
_%>
